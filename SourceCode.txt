fun classify(score: Double): String {
    return if (!score.isFinite() || score < 0 || score > 10) {
        "KhÃ´ng há»£p lá»‡"
    } else if (score >= 8) {
        "Giá»i"
    } else if (score >= 6.5) {
        "KhÃ¡"
    } else if (score >= 5) {
        "Trung bÃ¬nh"
    } else {
        "Yáº¿u"
    }
}

fun classificationColor(score: Double): Color {
    return if (!score.isFinite() || score < 0 || score > 10) {
        Gray
    } else if (score >= 8) {
        Color(0xFF387448)
    } else if (score >= 6.5) {
        Color(0xFF3865AE)
    } else if (score >= 5) {
        Color(0xFFA46A24)
    } else {
        Color(0xFFB84B45)
    }
}

fun formatNumber(number: Double): String {
    return if (number == 0.0) {
        "0"
    } else {
        java.math.BigDecimal.valueOf(number)
            .stripTrailingZeros()
            .toPlainString()
    }
}

@Composable
fun CalculatorScreen() {
    var firstInput by rememberSaveable { mutableStateOf("") }
    var secondInput by rememberSaveable { mutableStateOf("") }
    var operation by rememberSaveable { mutableStateOf("") }


    val first = firstInput.replace(',', '.').toDoubleOrNull()
    val second = secondInput.replace(',', '.').toDoubleOrNull()

    val result: String = if (operation.isEmpty()) {
        ""
    } else if (first == null || second == null) {
        "Nháº­p Ä‘á»§ hai sá»‘ há»£p lá»‡"
    } else if (!first.isFinite() || !second.isFinite()) {
        "Sá»‘ nháº­p khÃ´ng há»£p lá»‡"
    } else if (operation == "/" && second == 0.0) {
        "KhÃ´ng thá»ƒ chia cho 0"
    } else {
        val value = when (operation) {
            "+" -> first + second
            "-" -> first - second
            "*" -> first * second
            else -> first / second
        }

        if (value.isFinite()) {
            formatNumber(value)
        } else {
            "Káº¿t quáº£ vÆ°á»£t giá»›i háº¡n"
        }
    }

    val symbols = listOf("+", "-", "*", "/")
    val colors = listOf(
        Color(0xFFD94740),
        Color(0xFFDCA846),
        Color(0xFF623BDB),
        Color(0xFF222222)
    )

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.White)
            .safeDrawingPadding()
            .imePadding()
            .verticalScroll(rememberScrollState())
            .padding(24.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Spacer(Modifier.height(100.dp))

        Text(
            text = "Thá»±c hÃ nh 03",
            fontSize = 20.sp,
            fontWeight = FontWeight.Bold
        )

        Spacer(Modifier.height(60.dp))

        OutlinedTextField(
            value = firstInput,
            onValueChange = { firstInput = it },
            modifier = Modifier.fillMaxWidth(),
            placeholder = { Text("Nháº­p sá»‘ thá»© nháº¥t") },
            singleLine = true,
            shape = RoundedCornerShape(16.dp),
            keyboardOptions = KeyboardOptions(
                keyboardType = KeyboardType.Decimal
            )
        )

        Spacer(Modifier.height(24.dp))

        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            for (index in symbols.indices) {
                val symbol = symbols[index]

                Button(
                    onClick = { operation = symbol },
                    modifier = Modifier.size(56.dp),
                    shape = RoundedCornerShape(8.dp),
                    contentPadding = PaddingValues(0.dp),
                    colors = ButtonDefaults.buttonColors(
                        containerColor = colors[index]
                    ),
                    border = if (operation == symbol) {
                        BorderStroke(3.dp, Color.Black)
                    } else {
                        null
                    }
                ) {
                    Text(symbol, fontSize = 24.sp)
                }
            }
        }

        Spacer(Modifier.height(24.dp))

        OutlinedTextField(
            value = secondInput,
            onValueChange = { secondInput = it },
            modifier = Modifier.fillMaxWidth(),
            placeholder = { Text("Nháº­p sá»‘ thá»© hai") },
            singleLine = true,
            shape = RoundedCornerShape(16.dp),
            keyboardOptions = KeyboardOptions(
                keyboardType = KeyboardType.Decimal
            )
        )

        Spacer(Modifier.height(12.dp))

        Text(
            text = "Káº¿t quáº£: $result",
            modifier = Modifier.fillMaxWidth(),
            fontSize = 18.sp
        )
    }
}

@Composable
fun ScoreboardScreen() {
    val students = listOf(
        ScoreStudent("Nguyá»…n VÄƒn An", 9.0),
        ScoreStudent("Tráº§n Thá»‹ BÃ¬nh", 7.2),
        ScoreStudent("LÃª Minh Chi", null),
        ScoreStudent("Pháº¡m Quá»‘c DÅ©ng", 4.5),
        ScoreStudent("VÃµ Thanh Em", 5.8)
    )

    var total = 0.0
    var scoredCount = 0
    var excellentCount = 0
    var missingCount = 0
    var highest: Double? = null

    for (student in students) {
        val score = student.score

        if (score == null) {
            missingCount++
        } else if (score.isFinite() && score >= 0 && score <= 10) {
            total += score
            scoredCount++

            if (score >= 8) {
                excellentCount++
            }

            val currentHighest = highest
            if (currentHighest == null || score > currentHighest) {
                highest = score
            }
        }
    }

    val averageText = if (scoredCount > 0) {
        String.format(java.util.Locale.US, "%.2f", total / scoredCount)
    } else {
        "ChÆ°a cÃ³ Ä‘iá»ƒm"
    }

    ExerciseScreen("Báº£ng Ä‘iá»ƒm lá»›p") {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .background(Blue, RoundedCornerShape(14.dp))
                .padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            Text("Äiá»ƒm trung bÃ¬nh lá»›p", color = Color.White.copy(0.7f))

            Text(
                averageText,
                color = Color.White,
                fontSize = 38.sp,
                fontWeight = FontWeight.Bold
            )

            HorizontalDivider(color = Color.White.copy(alpha = 0.2f))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween
            ) {
                Statistic("Giá»i", "$excellentCount")
                Statistic("ChÆ°a cÃ³ Ä‘iá»ƒm", "$missingCount")
                Statistic("Cao nháº¥t", highest?.toString() ?: "â€”")
            }
        }

        Spacer(Modifier.height(6.dp))

        Text(
            "DANH SÃCH Â· ${students.size} SINH VIÃŠN",
            color = Gray,
            fontWeight = FontWeight.Bold,
            fontSize = 14.sp
        )

        for (student in students) {
            val score = student.score

            WhiteCard {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Column(modifier = Modifier.weight(1f)) {
                        Text(
                            student.name,
                            fontSize = 16.sp,
                            fontWeight = FontWeight.Bold
                        )

                        Spacer(Modifier.height(4.dp))

                        if (score == null) {
                            Text(
                                "ChÆ°a cÃ³ Ä‘iá»ƒm",
                                color = Gray,
                                fontSize = 13.sp
                            )
                        } else {
                            Text(
                                classify(score),
                                color = classificationColor(score),
                                fontSize = 13.sp
                            )
                        }
                    }

                    Text(
                        text = score?.toString() ?: "â€”",
                        fontSize = 22.sp,
                        fontWeight = FontWeight.Bold,
                        color = if (score == null) Gray else Color.Black
                    )
                }
            }
        }
    }
}

@Composable
fun BaiTapTuan2Theme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    // Dynamic color is available on Android 12+
    dynamicColor: Boolean = true,
    content: @Composable () -> Unit
) {
    val colorScheme = when {
        dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {
            val context = LocalContext.current
            if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
        }

        darkTheme -> DarkColorScheme
        else -> LightColorScheme
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = Typography,
        content = content
    )
}
